# SQL-практика: доставка банковских карт

## Схема базы данных

Сервис доставки банковских карт помогает клиентам получать новые карты и отслеживать заявки на доставку. В системе хранятся сведения о клиентах, их картах, заявках и курьерах, которые доставляют карты по указанным адресам. У каждой заявки есть дата оформления, статус, стоимость доставки и информация о том, завершена ли доставка.

```sql
CREATE TABLE clients (
  id SERIAL PRIMARY KEY,
  first_name VARCHAR(100) NOT NULL,
  last_name VARCHAR(100) NOT NULL,
  birth_date DATE NOT NULL,
  registration_date DATE NOT NULL,
  is_active BOOLEAN NOT NULL,
  email VARCHAR(255),
  bonus_balance NUMERIC(10,2) NOT NULL
);

CREATE TABLE cards (
  id SERIAL PRIMARY KEY,
  client_id INTEGER NOT NULL REFERENCES clients(id),
  card_number VARCHAR(20) NOT NULL,
  card_type VARCHAR(50) NOT NULL,
  issue_date DATE NOT NULL,
  is_active BOOLEAN NOT NULL,
  monthly_fee NUMERIC(10,2) NOT NULL
);

CREATE TABLE couriers (
  id SERIAL PRIMARY KEY,
  first_name VARCHAR(100) NOT NULL,
  last_name VARCHAR(100) NOT NULL,
  phone VARCHAR(20) NOT NULL,
  hire_date DATE NOT NULL,
  is_active BOOLEAN NOT NULL,
  rating NUMERIC(3,2)
);

CREATE TABLE delivery_orders (
  id SERIAL PRIMARY KEY,
  card_id INTEGER NOT NULL REFERENCES cards(id),
  courier_id INTEGER REFERENCES couriers(id),
  delivery_address VARCHAR(255) NOT NULL,
  order_date DATE NOT NULL,
  delivery_date DATE,
  status VARCHAR(30) NOT NULL,
  delivery_cost NUMERIC(10,2) NOT NULL,
  is_completed BOOLEAN NOT NULL,
  comment TEXT
);
```
